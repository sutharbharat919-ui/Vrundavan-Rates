<script>
// ડિફોલ્ટ ભાવો (જો ડિવાઇસ માં ડેટા સેવ ન હોય તો આ દેખાશે)
let rates = {
g24: +localStorage.getItem('g24') || 143480,
g22: +localStorage.getItem('g22') || 134400,
g18: +localStorage.getItem('g18') || 108730,
s95: +localStorage.getItem('s95') || 221260,
s90: +localStorage.getItem('s90') || 209240,
s60: +localStorage.getItem('s60') || 185185,
th_g999: +localStorage.getItem('th_g999') || 146300,
th_s999: +localStorage.getItem('th_s999') || 231250,
th_g22: +localStorage.getItem('th_g22') || 134600
};

// ડિસ્પ્લે પર ભાવો અપડેટ કરવા
function updateDisplay() {
// સ્ક્રીન પર દેખાડવા માટે
document.getElementById('d_g24').textContent = rates.g24.toLocaleString('en-IN');
document.getElementById('d_g22').textContent = rates.g22.toLocaleString('en-IN');
document.getElementById('d_g18').textContent = rates.g18.toLocaleString('en-IN');
document.getElementById('d_s95').textContent = rates.s95.toLocaleString('en-IN');
document.getElementById('d_s90').textContent = rates.s90.toLocaleString('en-IN');
document.getElementById('d_s60').textContent = rates.s60.toLocaleString('en-IN');
document.getElementById('d_th_g999').textContent = rates.th_g999.toLocaleString('en-IN');
document.getElementById('d_th_s999').textContent = rates.th_s999.toLocaleString('en-IN');
document.getElementById('d_th_g22').textContent = rates.th_g22.toLocaleString('en-IN');

// એડમિન પેનલ ના ઇનપુટ બોક્સ માં કિંમતો સેટ કરવી
document.getElementById('in_g24').value = rates.g24;
document.getElementById('in_g22').value = rates.g22;
document.getElementById('in_g18').value = rates.g18;
document.getElementById('in_s95').value = rates.s95;
document.getElementById('in_s90').value = rates.s90;
document.getElementById('in_s60').value = rates.s60;
document.getElementById('in_th_g999').value = rates.th_g999;
document.getElementById('in_th_s999').value = rates.th_s999;
document.getElementById('in_th_g22').value = rates.th_g22;

// તારીખ અને સમય સેટ કરવો
let savedTime = localStorage.getItem('update_time') || new Date().toLocaleTimeString('en-IN', {hour: '2-digit', minute:'2-digit'});
let savedDate = localStorage.getItem('update_date') || new Date().toLocaleDateString('en-IN');
document.getElementById('time-gold').textContent = savedTime;
document.getElementById('current-date').textContent = savedDate;
}

// એડમિન પેનલ શો / હાઇડ કરવા માટે
function toggleAdmin() {
let panel = document.getElementById('admin-panel');
if(panel.style.display === 'block') {
panel.style.display = 'none';
} else {
panel.style.display = 'block';
panel.scrollIntoView({ behavior: 'smooth' });
}
}

// નવા ભાવો સેવ કરવા માટે
function saveRates() {
rates.g24 = +document.getElementById('in_g24').value;
rates.g22 = +document.getElementById('in_g22').value;
rates.g18 = +document.getElementById('in_g18').value;
rates.s95 = +document.getElementById('in_s95').value;
rates.s90 = +document.getElementById('in_s90').value;
rates.s60 = +document.getElementById('in_s60').value;
rates.th_g999 = +document.getElementById('in_th_g999').value;
rates.th_s999 = +document.getElementById('in_th_s999').value;
rates.th_g22 = +document.getElementById('in_th_g22').value;

// LocalStorage માં ડેટા સ્ટોર કરવો
for (let key in rates) {
localStorage.setItem(key, rates[key]);
}

let now = new Date();
let timeStr = now.toLocaleTimeString('en-IN', {hour: '2-digit', minute:'2-digit'});
let dateStr = now.toLocaleDateString('en-IN');

localStorage.setItem('update_time', timeStr);
localStorage.setItem('update_date', dateStr);

updateDisplay();
alert('બધા ભાવો સફળતાપૂર્વક સેવ થઈ ગયા છે!');
toggleAdmin(); // પેનલ બંધ કરવા
}

// કેલ્ક્યુલેટર ફંક્શન (સોનું ૧૦ ગ્રામ અને ચાંદી ૧ કિલો (૧૦00 ગ્રામ) ના ભાવ મુજબ ગણતરી કરશે)
function calculatePrice() {
let item = document.getElementById('calc-item').value;
let weight = +document.getElementById('calc-weight').value;

if(!weight || weight <= 0) {
alert('કૃપા કરીને સાચું વજન લખો');
return;
}

let baseRate = rates[item];
let totalPrice = 0;

// જો આઈટમ ચાંદી (Silver) હોય તો ભાવ ૧ કિલો (૧૦૦૦ ગ્રામ) નો છે
if(item === 's95' || item === 's90' || item === 's60' || item === 'th_s999') {
totalPrice = (baseRate / 1000) * weight;
} else {
// સોના માટે ભાવ ૧૦ ગ્રામ નો છે
totalPrice = (baseRate / 10) * weight;
}

document.getElementById('calc-result').textContent = 'કુલ રકમ: ₹' + Math.round(totalPrice).toLocaleString('en-IN');
}

// એપ્લિકેશન શરૂ થતા જ ભાવ લોડ કરવા
updateDisplay();
